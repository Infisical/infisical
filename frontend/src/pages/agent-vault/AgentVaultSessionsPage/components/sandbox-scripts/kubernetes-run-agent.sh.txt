set -euo pipefail

INFISICAL_URL=https://app.infisical.com
# Logs in with Universal Auth. For Kubernetes Auth, call
# /api/v1/auth/kubernetes-auth/login with {"identityId", "jwt"} instead, where
# jwt is your orchestrator's service account token. The identity's allowed
# namespaces must include the orchestrator's namespace.
ACCESS_TOKEN=$(curl -fsS -X POST "$INFISICAL_URL/api/v1/auth/universal-auth/login" \
  -H "Content-Type: application/json" \
  -d "{\"clientId\": \"$INFISICAL_UNIVERSAL_AUTH_CLIENT_ID\", \"clientSecret\": \"$INFISICAL_UNIVERSAL_AUTH_CLIENT_SECRET\"}" \
  | jq -r .accessToken)
# Unique per run, so runs started at the same time don't share names.
RUN_ID=$(date +%s)-$(openssl rand -hex 3)
echo "Run ID: $RUN_ID"

# One session per run. The ttl outlasts the pod's 30-minute deadline, so the
# agent can't lose access mid-run.
SESSION=$(curl -fsS -X POST "$INFISICAL_URL/api/v1/agent-vault/sessions" \
  -H "Authorization: Bearer $ACCESS_TOKEN" \
  -H "Content-Type: application/json" \
  -d "{\"projectId\": \"<project-id>\", \"accessBundle\": \"<access-bundle-name>\", \"ttl\": \"35m\", \"label\": \"kubernetes run $RUN_ID\"}")
SESSION_ID=$(echo "$SESSION" | jq -r .session.id)

cleanup() {
  # Revoke first, so the session ends even if deleting the pod fails.
  curl -fsS -X POST "$INFISICAL_URL/api/v1/agent-vault/sessions/$SESSION_ID/revoke" \
    -H "Authorization: Bearer $ACCESS_TOKEN" > /dev/null \
    || echo "Couldn't revoke session $SESSION_ID. Revoke it in your Agent Vault project." >&2
  kubectl -n agents delete "pod/agent-$RUN_ID" "secret/agent-$RUN_ID" --ignore-not-found
}
trap cleanup EXIT
# Bash skips the EXIT trap when it's killed, such as when CI cancels a job.
trap 'exit 143' TERM INT

# The token goes through stdin, so it doesn't appear in the process list.
echo "$SESSION" | jq -r '"INFISICAL_AGENT_VAULT_SESSION_TOKEN=" + .session.token' \
  | kubectl -n agents create secret generic "agent-$RUN_ID" --from-env-file=/dev/stdin

sed "s/<run-id>/$RUN_ID/g" agent-pod.yaml | kubectl apply -f -

not_started() {
  echo "The agent's pod didn't start:" >&2
  kubectl -n agents get pod "agent-$RUN_ID" -o jsonpath='{.status.phase} {.status.containerStatuses[0].state.waiting.reason} {.status.containerStatuses[0].state.waiting.message}{.status.conditions[?(@.reason=="Unschedulable")].message}{"\n"}' >&2
  exit 1
}
# kubectl logs returns at once for a pod that isn't on a node yet, so wait for
# the scheduler first.
kubectl -n agents wait --for=condition=PodScheduled --timeout=2m "pod/agent-$RUN_ID" > /dev/null || not_started
# Wait up to 2 minutes for the container to start, then follow its output until
# the agent exits. In the background, so a canceled script cleans up right away
# instead of after the agent exits.
kubectl -n agents logs -f --pod-running-timeout=2m "agent-$RUN_ID" &
if ! wait $!; then
  # The container started, so kubectl stopped following because the script was canceled.
  [ -z "$(kubectl -n agents get pod "agent-$RUN_ID" -o jsonpath='{.status.containerStatuses[0].state.waiting}')" ] && exit 130
  not_started
fi

# Exit with the agent's exit code.
EXIT_CODE=""
until [ -n "$EXIT_CODE" ]; do
  sleep 1
  EXIT_CODE=$(kubectl -n agents get pod "agent-$RUN_ID" -o jsonpath='{.status.containerStatuses[0].state.terminated.exitCode}')
done
exit "$EXIT_CODE"
