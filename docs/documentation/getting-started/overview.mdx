  <Card title="Secrets Management" icon="vault" href="/documentation/platform/secrets-mgmt/overview">

  <Card title="Secrets Scanning" icon="radar" href="/documentation/platform/secret-scanning/overview">

  <Card title="Certificate Management" icon="file-certificate" href="/documentation/platform/pki/overview">
